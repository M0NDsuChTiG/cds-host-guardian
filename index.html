// LSM Hook: Блокировка несанкционированного повышения прав
SEC("lsm/task_fix_setuid")
int BPF_PROG(cds_restrict_setuid, struct cred *new, const struct cred *old) {
    u64 cg_id = bpf_get_current_cgroup_id();
    u32 *is_trusted = bpf_map_lookup_elem(&trusted_groups, &cg_id);

    // Zero Trust Check
    if (new->uid.val == 0 && (!is_trusted || *is_trusted != 1)) {
        bpf_printk("CDS: Privilege escalation blocked for cgroup %llu", cg_id);
        return -EPERM;
    }
    return 0;
}
